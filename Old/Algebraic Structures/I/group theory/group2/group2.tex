\begin{claim}{1a}
    Let $q:G \to G'$ be a surjective homomorphism. Prove that if G is cyclic, then G' is cyclic.
\end{claim}
\begin{proof}
    Let $q:G \to G'$ be a surjective homomorphism. Suppose G is a cyclic group. Then by definition of cyclic groups, there exists a $g \in G$ s.t $G = \ANG{g} = \{g^k : k \in \Z\}$. Since, q is surjective, it means for all $x \in G'$, $\exists y \in G$ s.t $q(y) = x$. By definition of a homomorphism, $q(g^k) = \prod_{i=1}^{k}q(g) = q(g)^k$ for all $k \in \Z$. Thus every element in $G'$ can be written as $q(g)^k$ for some $k \in \Z$. Thus let $h = q(g)$, since every element in $G'$ can be written as $h^k = q(g)^k$ for some $k \in \Z$, then $G' = \{h^k : k \in \Z\}$. Thus every element in G' can be generated by h. Thus by definition of cyclic groups, $G'$ is cyclic.
\end{proof}

\begin{claim}{1b}
    Let $q:G \to G'$ be a surjective homomorphism. If G is abelian, then G' is abelian.
\end{claim}
\begin{proof}
    Let $q:G \to G'$ be a surjective homomorphism. Suppose G is abelian. By definition of abelian groups, $\forall a, b \in G$ $ab = ba$. Thus $q(ab) = q(ba)$. By definition of homomorphisms, $q(ab) = q(a)q(b)$ and $q(ba) = q(b)q(a)$. Thus $q(a)q(b) = q(b)q(a)$. Since q is surjective, $\forall x, y \in G'$, there exists a $a, b \in G$ s.t $q(a) = x$ and $q(b) = y$. Now since $q(a)q(b) = q(b)q(a)$ for all $a, b \in G$, then $xy = yx$. Thus G' is commutative and thus is abelian.
\end{proof}

\begin{claim}{2a}
    The intersection $K \cap H$ of subgroups of a group G is a subgroup of H.
\end{claim}
\begin{proof}
    Suppose $K$, and $H$ are subgroups of G. By definition of a intersection, $K \cap H \subseteq H$.

    \begin{enumerate}{}
        \item \textbf{Identity: } $e \in G$ by the identity axiom of groups and H and K have a identity element by the same. Since the identity element is unique in a group and the identity element of any subgroup is the identity element of the parent group, $e \in K$ and $e \in H$. Thus $e \in K \cap H$.
        \item \textbf{Inverse: } For any $a \in K \cap H$, since $K$ and $H$ are subgroups then by the inverse axiom $a^{-1} \in K$ and $a^{-1} \in H$. Thus $a^{-1} \in K \cap H$
        \item \textbf{Closure: } $\forall a, b \in K \cap H$, $ab \in K$ and $ab \in H$ by the closure property of groups. Thus $ab \in K \cap H$.
    \end{enumerate}

    Thus $K \cap H$ is a subgroup of H.
\end{proof}

\begin{claim}{2b}
    If K is a normal subgroup of G, then $K \cap H$ is a normal subgroup of H.
\end{claim}
\begin{proof}
    Suppose K is a normal subgroup of G. By definition of a intersection, $\forall a \in K \cap H$, $a \in K$ and $a \in H$. By definition of a subgroup, $\forall h \in H$, $h \in G$. By definition of a normal subgroups, $hah^{-1} \in K$. Furthermore since $H$ is closed under the binary operation, $hah^{-1} \in H$. Thus $hah^{-1} \in K \cap H$ $\forall a \in K \cap H$ and $\forall h \in H$. Thus $K \cap H$ is a normal subgroup.
\end{proof}

\begin{claim}{3a}
    Let U denote the group of invertible upper triangular $2 \times 2$ matrices $A = \begin{bmatrix}{}
        a & b \\
        0 & c
    \end{bmatrix}$, and $\phi: U \to \R^\times$ be a map which sends $A \to a^2$. Prove that $\phi$ is a homomorphism.
\end{claim}
\begin{proof}
    Let U denote the group of invertible upper triangular $2 \times 2$ matrices $A = \begin{bmatrix}{}
        a & b \\
        0 & c
    \end{bmatrix}$, and $\phi: U \to \R^\times$ be a map which sends $A \to a^2$. For all $A, B \in U$. A and B can be written by definition of U as the following $A = \begin{bmatrix}{}
        a_1 & b_1 \\
        0 & c_1
    \end{bmatrix}$ and $B = \begin{bmatrix}{}
        a_2 & b_2 \\
        0 & c_2
    \end{bmatrix}$ $\forall a_1, b_1, c_1, a_2, b_2, c_2 \in \R$. $AB = \begin{bmatrix}{}
        a_1a_2 & a_1b_2 + b_1c_1 \\
        0 & b_1b_2 + c_1c_2
    \end{bmatrix}$. $\phi(AB) = a_1a_2$. $\phi(A) = a_1$ and $\phi(B) = a_2$. Thus for all $A,B \in U$, $\phi(AB) = a_1a_2 = \phi(A)\phi(B)$. Thus by definition of a homomorphism, $\phi$ is a homomorphism.
\end{proof}

\begin{PROB}{3b}\label{Problem 3b.}
    Determine $\phi$'s kernel and image.
\end{PROB}
\begin{solution}
    By definition of kernel of a homomorphism, its all the elements in the domain which map to 1. By definition of $\phi$, which maps any invertible upper triangular matrix to the square of its top left corner element, its all invertible upper triangular matrices with $\pm 1$ on the top left corner. Thus any matrix of the form $\begin{bmatrix}{}
        \pm 1 & b \\
        0 & c
    \end{bmatrix}$ $\forall b, c \in \R$ s.t $c \neq 0$. The $\phi_*(U) = \R$, because for each $a \in \R$ there exists a matrix $A = \begin{bmatrix}{}
        \sqrt{a} & 0 \\
        0 & 1
    \end{bmatrix} \in U$ s.t $\phi(A) = a$. Thus the image of $\phi$ is $\R_+$(Positive reals) and $\phi$ is surjective.
\end{solution}

\begin{claim}{4a}
    Let $f :\R^+ \to \C^\times$ be the map $f(x) = e^{ix}$. $f$ is a homomorphism.
\end{claim}
\begin{proof}
    Let $f :\R^+ \to \C^\times$ be the map $f(x) = e^{ix}$. For all $a, b \in \R^+$, $f(a) = e^{ia}$ and $f(b)=e^{ib}$ and $f(a)f(b) = e^{ia}e^{ib} = e^{i(a + b)}$ (properties of exponents). $f(a+b) = e^{i(a + b)} = f(a)f(b)$. Thus by definition of a homomorphism, $f$ is a homomorphism.
\end{proof}

\begin{PROB}{4b}\label{Problem 4b.}
    Determine $f$'s kernel and image.
\end{PROB}
\begin{solution}
    By definition of kernel of a homomorphism, its all the elements in the domain which map to 1. By definition of 1, $1 = e^{i2k\pi}$ for all $k \in \Z$. Thus by definition of f, $\ker f = \{2k\pi : k \in \Z\}$. The image of f, is any element on the unit circle in the complex plane since complex exponentiation is the same as rotation in the complex plane.
\end{solution}

\begin{claim}{5a}
    Let H be the set of $n \times n$ matrices that have the block form $M = \begin{bmatrix*}{}
        A & B \\
        0 & D
    \end{bmatrix*}$ with $A \in GL_r(\R)$ and D in $GL_{n-r}(\R)$, form a subgroup H of $GL_n(\R)$.
\end{claim}
\begin{proof}
    Let H be the set of $n \times n$ matrices that have the block form $M = \begin{bmatrix*}{}
        A & B \\
        0 & D
    \end{bmatrix*}$ with $A \in GL_r(\R)$ and D in $GL_{n-r}(\R)$. 

    For all $X \in H$, $X = \begin{bmatrix*}{}
        A & B \\
        0 & D
    \end{bmatrix*}$ for $A \in GL_n(\R)$ and $B \in GL_{(n-r)}(\R)$. $\det(X) = \det(A)\det(D)$, since $\det(A) \neq 0$ and $\det(D) \neq 0$ because A and B are invertible. Thus $\det(X) \neq 0$. Thus $X$ is invertible. $H \subset GL_n(\R)$.

    \begin{enumerate}{}
        \item \textbf{Identity: } $I_n \in H$, since $A = I_r$, $B = 0$, and $D = I_{n-r}$. Thus $H$ has the identity.
        \item \textbf{Inverse: } By definition since $H \subset GL_n(\R)$, then each element $H$ is invertible thus there exists a $N \in GL_n(\R)$. $N = \begin{bmatrix*}
            A^{-1} & -A^{-1}BD^{-1} \\
            0 & D^{-1}
        \end{bmatrix*}$. $MN = \begin{bmatrix}{}
            AA^{-1} & -AA^{-1}BD^{-1} + BD^{-1} \\
            0 & DD^-1
        \end{bmatrix} = \begin{bmatrix}{}
            I_{r} & 0 \\
            0 & I_{n-r} = I_n
        \end{bmatrix} $. $NM = \begin{bmatrix}{}
            AA^{-1} & A^{-1}B - A^{-1}BDD^{-1} \\
            0 & DD^-1
        \end{bmatrix} = \begin{bmatrix}{}
            I_{r} & 0 \\
            0 & I_{n-r}
        \end{bmatrix} = I_n$. Thus every M has a N inverse. 
        \item \textbf{Closure: } For all $X, Y \in H$. $X = \begin{bmatrix}
            A_1 & B_1 \\
            0 & C_1
        \end{bmatrix}$ and $Y = \begin{bmatrix}
            A_2 & B_2 \\
            0 & C_2
        \end{bmatrix}$. $XY = \begin{bmatrix}
            A_1 & A_1B_2 + C_1C_2 \\
            0 & C_1C_2
        \end{bmatrix} \in H$. Thus H is closed.
    \end{enumerate}

    Thus H is a subgroup of $GL_n(\R)$.
\end{proof}

\begin{claim}{5b}
   The map $H \to GL_r(\R)$ that sends $M\to A$ is a homomorphism.
\end{claim}
\begin{proof}
    Let $f: H \to GL_r(\R)$ be the map that sends $M\to A$. For all $X, Y \in H$. $X = \begin{bmatrix}
        A_1 & B_1 \\
        0 & C_1
    \end{bmatrix}$ and $Y = \begin{bmatrix}
        A_2 & B_2 \\
        0 & C_2
    \end{bmatrix}$. $XY = \begin{bmatrix}
        A_1 & A_1B_2 + C_1C_2 \\
        0 & C_1C_2
    \end{bmatrix}$. $f(X) = A_1$ and $f(Y) = A_2$. $f(X)f(Y) = A_1A_2$. $f(XY) = A_1A_2$. Thus $f(XY) = f(X)f(Y)$. Thus by definition of homomorphism, f is homomorphism.
\end{proof}

\begin{PROB}{5c}\label{Problem 5c.}
    What is the kernel of the map?
\end{PROB}
\begin{solution}
    The kernel of $f$ is all the matrices in H of the form $M = \begin{bmatrix}
        I_r & B \\
        0 & D
    \end{bmatrix}$ for all $D \in GL_{n-r}(\R)$. Thus $f(M) = I_r$.
\end{solution}

\begin{PROB}{6}\label{Problem 6.}
    Determine the center of $Gl_n(\R)$.
\end{PROB}
\begin{claim}{6a}
    Any matrix in the the center commutes with a off diagonal matrix
\end{claim}
\begin{proof}
    Let $e_{i, j}(k)$ be the matrix of 0s with $a_{i, j} = k$. $e_{i, j}(k)$ is a off diagonal matrix.
    Every matrix in the center commutes with every element of $GL_n(\R)$ by definition. Thus any matrix in the center commutes with a off diagonal matrix $e_{i, j}(k)$
\end{proof}
\begin{claim}{6b}       
    Any matrix which commutes with a off diagonal matrix is diagonal.
\end{claim}
\begin{proof}
    Assume the contrary, $A$ is not a diagonal matrix, s.t exists a $a_{i, j} \neq 0$ where $i \neq j$ which commutes with $B = e_{i, j}(1)$. Left multiplying by B is the same as taking the jth row and making it the ith row while putting a 0 everywhere else. Right multiplying by B is the same as taking the ith column and making it the jth column. Thus $a_{j,j} \neq 0$ of $AB$ while $a_{j, j} = 0$ of $BA$. Thus $BA \neq AB$. Thus by contradiction, any matrix commutes with a off diagonal matrix has no off diagonal elements thus it is diagonal.
\end{proof}

Thus all elements of the center are diagonal.

\begin{claim}{6c}
    Any diagonal matrix which commutes with all permutation matrices is scalar
\end{claim}
\begin{proof}
    Assume the contrary, A is a diagonal matrix which is not scalar but commutes with all permutation matrices. Thus there exists i and j s.t $a_{ii} \neq a_{jj}$. Let $P_{i,j}$ be the permutation matrix which when right multiplied swaps the $i$th and $j$th columns. However when left multiplying, $P_{i, j}$ swaps the $i$th and $j$th rows. $(ab)_{i,j} = a_{i,i}$ of $AB$ while $(ba)_{i,j} = a_{j,j}$ of $BA$. Thus $AB \neq BA$. Thus by contradiction A must be scalar.
\end{proof}

Thus all elements of the center must be scalar matrices.
