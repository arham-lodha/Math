\section*{Topological Primer}

\begin{definition}
    A \textbf{topological group} G is a group $G$ equipped with the topology such $(x, y) \to xy$ and $i: x \to x^{-1}$ are continous. $\forall g \in G$, $L_g : G \to G$ is left multiplication by $g$.     
\end{definition}

\begin{definition}
    A \textbf{symmetric} neighborhood $V$ is a neighborhood such that $V = V^{-1}$.   
\end{definition}

\begin{claim}{2.1}\label{Claim 2.1.}
    Let $G$ be a topological group: 
    
    \begin{enumerate}
        \item The topology of $G$ is invariant under translations and inversions, that is $\forall U \in \T_G$, $\forall g \in G$ $gU, Ug, U^{-1} \in \T_G$. Furthermore $\forall A \subeq G$, $AU, UA \in \T_G$
        \item $\forall U \in \T \ni 1 \in U$, $\exists V$ symmetric such that $V V \subeq U$  
        \item If $H \leq G \implies \overline{H} \leq G$ 
        \item Every open subgroup of G is closed
        \item If A and B are compact sets in G, so is AB    
    \end{enumerate}
    
\end{claim}
\begin{proof}
    \begin{enumerate}
        \item $L_{g^{-1}}^*(U) = gU$, $L_{g}^*(U) = g^{-1}U$, $i^*(U) = U^{-1}$ are all open by continuity. $AU = \bigcup_{a \in A} aU$ and $UA = \bigcup_{a \in A} Ua$ are open by closure of unions in topology. 
        \item Continuity of $(x, y) \to xy$ means that $\exists W_1, W_2 \in \T \ni 1 \in W_1 \cap W_2, W_1 W_2 \subeq U$. Take $V = W_1 \cap W_2 \cap W_1^{-1} \cap W_2^{-1}$.
        \item Just prove inverses and closure of multiplication exist. $\forall x \in \overline{H} \setminus H, \exists U_x \in \T \ni x \in H \cap U_x$         
    \end{enumerate}
    
\end{proof}

\begin{definition}
    $G$ is a \textbf{locally compact group} if and only if $G$ is hausdorff and locally compact (every point has a compact neighborhood).   
\end{definition}

\section*{Haar Measure}

\begin{definition}
    A \textbf{left Haar Measure} on G is a nonzero Radon measure (???) $\mu$ on G such that $\mu(xE) = \mu(E)$ for every Borel set $E \subeq G$ and $\forall x \in G$    
\end{definition}

\begin{claim}{2.9}\label{Claim 2.9.}
    Let $\mu$ be a Radon measure on a locally compact group $G$, and let $\tilde{\mu}(E) = \mu(E^{-1})$
    \begin{enumerate}
        \item $\mu$ is a left Haar measure $\iff$ $\tilde{\mu}$ is a right Haar measure  
        \item  µ is a left Haar measure if and only if $\int L_y f d \mu = \int f d \mu$  for every
        $f \in C^+_c(G)$ and every $\forall y \in G$.
    \end{enumerate}
\end{claim}
\begin{proof}
    \begin{enumerate}
        \item $\implies$ Suppose $\mu$ is a left Haar measure. $\forall E \subeq G$, borel, and $\forall x \in G$, $\tilde{\mu}(Ex) = \mu((Ex)^{-1}) = \mu(x^{-1}E^{-1}) = \mu(E^{-1}) = \tilde{\mu}(E^{-1})$.
        $\impliedby$: Reverse the last thing.   
        \item $\implies$: Suppose $\mu$ is a Haar measure, then $\int L_y f d \mu = \int f d \mu_y = \int f d \mu$ for all $f$, by approximating $f$ by simple functions. 
        $\impliedby:$ Suppose $\forall f \in C_c^+(G)$, $\int L_y f d \mu = \int f d \mu_y = \int f d \mu$, by uniqueness of Riesz representation $\mu = \mu_y$.            
    \end{enumerate}
      
\end{proof}
